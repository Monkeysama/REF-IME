cmake_minimum_required(VERSION 3.25)
set(REF_IME_VERSION "1.0.0" CACHE STRING "Plugin release version supplied by a Git tag")
if(NOT REF_IME_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
    message(FATAL_ERROR "REF_IME_VERSION must have the form N.N.N")
endif()
project(REF_IME VERSION ${REF_IME_VERSION} LANGUAGES CXX)
add_compile_definitions(REF_IME_VERSION="${PROJECT_VERSION}")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "REF-IME requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(REF_IME SHARED src/plugin.cpp src/host_api.cpp src/ime_proxy.cpp)
set_target_properties(REF_IME PROPERTIES OUTPUT_NAME "REF-IME")
target_include_directories(REF_IME PRIVATE src .deps/imgui .deps/reframework)
target_compile_definitions(REF_IME PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN IMGUI_USE_WCHAR32 IMGUI_ENABLE_FREETYPE)
target_compile_options(REF_IME PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(REF_IME PRIVATE user32 imm32 gdi32)
